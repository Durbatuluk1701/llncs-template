\subsection{Tables, Equations and Figures}

\Cref{tab:sizes} uses \texttt{booktabs}: no vertical rules and only three
horizontal ones.  Table captions go above the table.

\begin{table}[tbp]
  \caption{A sample table.}
  \label{tab:sizes}
  \centering
  \begin{tabular}{@{}lr@{}}
    \toprule
    Component        & Lines of code \\
    \midrule
    Specification    & 1\,200 \\
    Proofs           & 8\,700 \\
    \bottomrule
  \end{tabular}
\end{table}

Displayed equations are centered and numbered, as in \cref{eq:sum}:
\begin{equation}
  \label{eq:sum}
  \sum_{i=1}^{n} i = \frac{n(n+1)}{2}.
\end{equation}

Figure captions go below the figure (\cref{fig:example}).  Prefer vector
graphics (PDF) to raster images.

\begin{figure}[tbp]
  \centering
  \includegraphics[width=0.6\linewidth]{example-image}
  \Description{A gray placeholder image.}  % alt text, for screen readers
  \caption{A sample figure.}
  \label{fig:example}
\end{figure}

\subsection{Theorems and Proofs}

The environments \texttt{theorem}, \texttt{lemma}, \texttt{corollary},
\texttt{proposition}, \texttt{conjecture}, \texttt{definition},
\texttt{example}, \texttt{remark} and \texttt{proof} exist in every style.

\begin{definition}[Successor]
  \label{def:succ}
  The successor of a natural number $n$ is $n + 1$.
\end{definition}

\begin{theorem}
  \label{thm:succ}
  Every natural number has a successor.
\end{theorem}

\begin{proof}
  Immediate from \cref{def:succ}.
\end{proof}
